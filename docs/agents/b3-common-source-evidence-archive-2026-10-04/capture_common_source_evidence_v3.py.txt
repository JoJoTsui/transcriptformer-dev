"""Versioned capture with explicit original-full-run collection reconciliation.

This ignored collector reads bounded JSON/XML, hashes declared source and
metadata files, and asks Git for tracked baseline identities. It never runs the
application, imports project modules, or opens H5/numeric payloads. Large stage
artifacts retain the producing stage's exact byte references without a second
payload read; that limitation is explicit in the output.

The original full pytest passed 1092 cases and skipped five, while its harness
returned one for requiring nonempty tests in two CLI utility files. This
version requires a separate source-bound collect-only reconciliation and
preserves every original failed outer status. It never rewrites raw receipts
or labels the collection-only invocation as a new full test execution.
"""

def require(condition: bool, message: str) -> None:
    if not condition:
        raise ValueError(message)


def bounded_metadata_bytes(path: Path, cap: int) -> bytes:
    require(path.suffix.lower() in METADATA_SUFFIXES, f"Payload is not metadata: {path}")
    initial = path.stat()
    require(stat.S_ISREG(initial.st_mode) and 0 <= initial.st_size <= cap,
            f"Missing/nonregular/oversize metadata: {path}")
    descriptor = os.open(path, os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW)
    with os.fdopen(descriptor, "rb") as stream:
        opened = os.fstat(stream.fileno())
        require(stat.S_ISREG(opened.st_mode)
                and (opened.st_dev, opened.st_ino, opened.st_size)
                == (initial.st_dev, initial.st_ino, initial.st_size)
                and opened.st_size <= cap, f"Metadata changed before bounded read: {path}")
        pieces, total = [], 0
        while True:
            piece = stream.read(min(1024**2, cap - total + 1))
            if not piece:
                break
            total += len(piece)
            require(total <= cap, f"Metadata grew beyond cap: {path}")
            pieces.append(piece)
        finished = os.fstat(stream.fileno())
        require((finished.st_dev, finished.st_ino, finished.st_size)
                == (opened.st_dev, opened.st_ino, opened.st_size)
                and total == opened.st_size, f"Metadata changed during bounded read: {path}")
    final = path.stat()
    require((final.st_dev, final.st_ino, final.st_size)
            == (opened.st_dev, opened.st_ino, opened.st_size),
            f"Metadata path changed after bounded read: {path}")
    return b"".join(pieces)


def read_small(path: Path, cap: int = METADATA_CAP) -> bytes:
    require(type(cap) is int and 0 < cap <= METADATA_CAP, "Invalid metadata cap")
    caller = os.path.abspath(os.fspath(path))
    canonical = Path(caller).resolve(strict=True)
    previous_resolution = CALLER_RESOLUTIONS.get(caller)
    require(previous_resolution is None or previous_resolution == str(canonical),
            f"Metadata caller path redirected after first admission: {caller}")
    require(str(canonical.resolve(strict=True)) == str(canonical),
            f"Canonical metadata path redirected before admission: {canonical}")
    data = bounded_metadata_bytes(canonical, cap)
    require(str(Path(caller).resolve(strict=True)) == str(canonical)
            and str(canonical.resolve(strict=True)) == str(canonical),
            f"Metadata path redirected during admission: {caller}")
    ref = {"path": str(canonical), "sha256": sha256(data).hexdigest(), "bytes": len(data)}
    prior = PINNED_REFS.get(str(canonical))
    require(prior is None or prior == ref, f"Pinned metadata/source bytes changed: {canonical}")
    CALLER_RESOLUTIONS.setdefault(caller, str(canonical))
    PINNED_REFS.setdefault(str(canonical), ref)
    return data


def load(path: Path) -> dict:
    def unique(pairs: list[tuple[str, object]]) -> dict:
        result = {}
        for key, value in pairs:
            require(key not in result, f"Duplicate JSON key: {key}")
            result[key] = value
        return result

    value = json.loads(read_small(path), object_pairs_hook=unique,
                       parse_constant=lambda x: (_ for _ in ()).throw(ValueError(x)))
    require(isinstance(value, dict), f"JSON object required: {path}")
    return value


def small_ref(path: Path) -> dict:
    caller = os.path.abspath(os.fspath(path))
    read_small(path)
    canonical = CALLER_RESOLUTIONS[caller]
    return PINNED_REFS[canonical].copy()


def recheck_pins() -> None:
    original_pins = {name: ref.copy() for name, ref in PINNED_REFS.items()}
    original_callers = CALLER_RESOLUTIONS.copy()
    for caller, original in original_callers.items():
        require(str(Path(caller).resolve(strict=True)) == original,
                f"Metadata caller alias redirected at final seal: {caller}")
    for name, first in original_pins.items():
        path = Path(name)
        require(str(path.resolve(strict=True)) == name,
                f"Canonical metadata path redirected at final seal: {name}")
        data = bounded_metadata_bytes(path, METADATA_CAP)
        require(str(path.resolve(strict=True)) == name,
                f"Canonical metadata path redirected during final seal: {name}")
        fresh = {"path": name, "sha256": sha256(data).hexdigest(), "bytes": len(data)}
        require(fresh == first, f"Pinned file bytes changed at final seal: {name}")
    for caller, original in original_callers.items():
        require(str(Path(caller).resolve(strict=True)) == original,
                f"Metadata caller alias redirected during final seal: {caller}")
    require(PINNED_REFS == original_pins and CALLER_RESOLUTIONS == original_callers,
            "Metadata pin registry changed during final seal")


def exact_ref(value: dict, *, metadata: bool = False) -> dict:
    require(isinstance(value, dict) and set(value) == {"path", "sha256", "bytes"}, "Bad byte reference")
    path = Path(value["path"])
    require(path.is_absolute() and str(path.resolve(strict=True)) == value["path"], "Noncanonical byte path")
    require(type(value["bytes"]) is int and value["bytes"] >= 0, "Bad byte count")
    require(isinstance(value["sha256"], str) and len(value["sha256"]) == 64
            and all(x in "0123456789abcdef" for x in value["sha256"]), "Bad SHA256")
    require(path.is_file() and path.stat().st_size == value["bytes"], "Referenced file size changed")
    if metadata:
        require(small_ref(path) == value, "Metadata bytes changed")
    return value


def sealed_ref(value: dict) -> dict:
    """Rehash allowlisted bounded metadata; retain producer seal for all payloads."""
    exact_ref(value)
    path = Path(value["path"])
    verified = path.suffix.lower() in METADATA_SUFFIXES and value["bytes"] <= METADATA_CAP
    if verified:
        require(small_ref(path) == value, "Metadata bytes changed")
    return {"reference": value, "collector_rehashed": verified,
            "verification": "sha256_and_size" if verified else "producer_sha256_shape_and_current_size_only"}


def git(*args: str) -> bytes:
    return subprocess.check_output(["git", *args], cwd=ROOT)


def tree_entries(revision: str) -> dict[str, tuple[str, str]]:
    result = {}
    for item in git("ls-tree", "-r", "-z", revision).split(b"\0"):
        if not item:
            continue
        header, name = item.split(b"\t", 1)
        mode, kind, blob = header.decode().split()
        require(kind == "blob", "Unexpected Git tree object")
        result[name.decode()] = (mode, blob)
    return result


def index_entries() -> dict[str, tuple[str, str]]:
    result = {}
    for item in git("ls-files", "-s", "-z").split(b"\0"):
        if not item:
            continue
        header, name = item.split(b"\t", 1)
        mode, blob, stage = header.decode().split()
        require(stage == "0", "Unmerged Git index")
        result[name.decode()] = (mode, blob)
    return result


def git_python_inventory(head: str) -> tuple[dict, dict, dict]:
    index = {p: {"mode": m, "blob": b} for p, (m, b) in index_entries().items()
             if p.endswith(".py")}
    tree = {p: {"mode": m, "blob": b} for p, (m, b) in tree_entries(head).items()
            if p.endswith(".py")}
    require(index == tree and len(index) == 233, "Current 233 Python index/HEAD inventory differs")
    digests = {p: small_ref(ROOT / p)["sha256"] for p in index}
    return index, tree, digests


def relative_name(name: str) -> str:
    require(isinstance(name, str) and not Path(name).is_absolute(), "Nonrelative Git path")
    require(str((ROOT / name).resolve(strict=True)).startswith(str(ROOT.resolve()) + os.sep),
            "Git path leaves repository")
    return name


def preserved_map(values: dict, count: int, tree: dict, index: dict, label: str) -> dict:
    require(isinstance(values, dict) and len(values) == count, f"{label} inventory count differs")
    for name, digest in values.items():
        relative_name(name)
        require(tree.get(name) is not None and index.get(name) == tree[name],
                f"{label} Git mode/blob differs: {name}")
        require(small_ref(ROOT / name)["sha256"] == digest, f"{label} bytes differ: {name}")
    return {"count": count, "baseline": label,
            "git_mode_blob_and_worktree_sha256_verified": True}


def source_audit(freeze: dict, expected: dict,
                 full_index: dict[str, tuple[str, str]]) -> dict:
    old = load(OLD_AUDIT)
    require(old.get("baseline") == ORIGINAL_BASELINE, "Backend audit baseline differs")
    require(old["preserved_file_counts"] == {"python": 224, "json": 98, "jsonl": 2}, "Old count audit differs")
    require(old["native_modules"] == 58 and old["original_pilot_software"] == {"human": 107, "mouse": 109},
            "Original native/pilot source counts differ")
    require(old["existing_bytes_and_git_index_modes_unchanged"] is True, "Old audit did not pass")
    old_source = freeze["original_source_file_sha256"]
    original_native = freeze["original_native_consumer_file_sha256"]
    original_app = freeze["legacy_application_consumer_file_sha256"]
    current_native = freeze["batch_consumer_file_sha256"]
    current_app = freeze["application_consumer_file_sha256"]
    require((len(original_native), len(original_app), len(current_native), len(current_app)) == (67, 74, 69, 82),
            "Consumer closure counts differ")
    require(all(current_native.get(p) == h for p, h in original_native.items()), "Original 67 altered")
    require(all(current_app.get(p) == h for p, h in original_app.items()), "Original 74 altered")
    new_paths = {str((ROOT / p).resolve()) for p in SOURCE_NAMES}
    require(set(current_native) - set(original_native) == {
        str((ROOT / SOURCE_NAMES[0]).resolve()), str((ROOT / SOURCE_NAMES[2]).resolve())},
        "Native closure has unexpected additions")
    require(set(current_app) - set(original_app) == new_paths | set(freeze["comparator_transitive_file_sha256"]),
            "Application closure has unexpected additions")
    require(set(freeze["comparator_transitive_file_sha256"]) <= set(old_source),
            "Comparator transitive absent from original freeze")
    require(len(set(old_source) & new_paths) == 0, "New scripts overlap historical source set")
    require(all(small_ref(Path(p))["sha256"] == h for p, h in old_source.items()), "Historical source bytes changed")
    require(all(small_ref(Path(p))["sha256"] == h for p, h in current_native.items()), "Native source bytes changed")
    require(all(small_ref(Path(p))["sha256"] == h for p, h in current_app.items()), "Application source bytes changed")
    require({p: small_ref(ROOT / p)["sha256"] for p in SOURCE_NAMES} == expected,
            "Independently reviewed source SHAs differ")
    original_ref = old["original_audit"]
    require(original_ref["path"] == str(OLD_AUDIT.parent.relative_to(ROOT) /
                                        "paged_native_cache_source_audit.json"), "Original audit path differs")
    original_path = ROOT / original_ref["path"]
    require(small_ref(original_path)["sha256"] == original_ref["sha256"]
            and small_ref(original_path)["bytes"] == original_ref["bytes"],
            "Original audit reference differs")
    original = load(original_path)
    require(original.get("baseline") == ORIGINAL_BASELINE
            and original.get("preserved_file_counts") == {"python": 224, "json": 98, "jsonl": 2},
            "Original audit baseline/counts differ")
    index = full_index
    original_tree = tree_entries(ORIGINAL_BASELINE)
    original_checks = {kind: preserved_map(original["preserved"][kind], number,
                                           original_tree, index, ORIGINAL_BASELINE)
                       for kind, number in (("python", 224), ("json", 98), ("jsonl", 2))}
    complete_tree = tree_entries(COMPLETE_BASELINE)
    complete_python = old["complete_regression_python_file_sha256"]
    complete_check = preserved_map(complete_python, 228, complete_tree, index, COMPLETE_BASELINE)
    current_tree = tree_entries("HEAD")
    require(all(current_tree.get(p) == index.get(p) for p in SOURCE_NAMES), "New source index/HEAD differs")
    counts = {"python": 0, "json": 0, "jsonl": 0}
    for name in index:
        for kind, suffix in (("python", ".py"), ("json", ".json"), ("jsonl", ".jsonl")):
            if name.endswith(suffix):
                counts[kind] += 1
    return {"historical_audit": small_ref(OLD_AUDIT), "original_audit": small_ref(original_path),
            "original_baseline": ORIGINAL_BASELINE, "complete_regression_baseline": COMPLETE_BASELINE,
            "original_preserved_inventories": original_checks,
            "complete_regression_python_inventory": complete_check,
            "historical_native_modules": 58, "original_pilot_software": {"human": 107, "mouse": 109},
            "original_consumer_counts": {"native": 67, "application": 74},
            "new_consumer_counts": {"native": 69, "application": 82},
            "current_git_index_suffix_counts": counts,
            "source_sha256": expected}


def junit(path: Path) -> dict:
    data = read_small(path)
    root = ET.fromstring(data)
    suites = [root] if root.tag == "testsuite" else list(root.iter("testsuite"))
    require(bool(suites), "JUnit has no suites")
    counts = {k: sum(int(s.get(k, "0")) for s in suites) for k in ("tests", "failures", "errors", "skipped")}
    counts["passed"] = counts["tests"] - counts["failures"] - counts["errors"] - counts["skipped"]
    require(counts["passed"] >= 0, "Invalid JUnit counts")
    cases = []
    for item in root.iter("testcase"):
        cases.append({"classname": item.get("classname"), "name": item.get("name"),
                      "passed": not any(child.tag in {"failure", "error", "skipped"}
                                        for child in item)})
    require(len(cases) == counts["tests"], "JUnit suite count/case count differs")
    return {"artifact": small_ref(path), "counts": counts, "cases": cases,
            "junit_suite_seconds": sum(float(s.get("time", "0")) for s in suites)}


def selected_cases(report: dict) -> dict:
    selected = {"native": [], "application": []}
    for case in report["cases"]:
        name = case["classname"] or ""
        side = ("native" if name.endswith("test_b3_paged_native_common_source")
                else "application" if name.endswith("test_b3_paged_native_common_bootstrap")
                else None)
        if side:
            selected[side].append(case)
    for side, total in (("native", 45), ("application", 30)):
        rows = selected[side]
        ids = {(x["classname"], x["name"]) for x in rows}
        require(len(rows) == len(ids) == total and all(x["passed"] for x in rows),
                f"Final suite does not prove {total} distinct passing {side} cases")
    return selected

            and marker["whole_method_fit_established"] is False and marker["scientific_readiness"] == "unavailable",
            "Study overstates readiness")

    freeze_marker = load(path / "stages/00-freeze/summary.json")
    binding_ref = freeze_marker["outputs"]["binding"]
    exact_ref(binding_ref, metadata=True)
    freeze = load(Path(binding_ref["path"]))

        "max_cooperative_stage_seconds", "max_supervised_stage_seconds",

        "disk_free_floor_bytes": 20 * 1024**3, "numeric_cap_bytes": 200 * 1024**2},
        "Frozen host/cost limits differ")

    for (name, action), job in zip(STAGES, marker["jobs"], strict=True):
        require(job["name"] == name and job["action"] == action and job["controller_return_code"] == 0,

        summary = load(summary_path)

            require(summary.get("binding") == binding_ref, "Stage is bound to another source freeze")
        require(exact_ref(job["summary"], metadata=True) == small_ref(summary_path), "Stage marker bytes differ")

            sealed_ref(item)
        available = {item["path"]: item for item in [*summary["inputs"], *summary["generated_files"]]}
        for item in summary["outputs"].values():
            if isinstance(item, dict) and set(item) == {"path", "sha256", "bytes"}:
                require(item == available.get(item["path"]), "Stage output is not in its sealed byte set")
        state = load(Path(job["supervisor_state"]["path"]))
        require(state.get("status") == "completed" and state.get("return_code") == 0,
                "Supervisor did not complete")
        exact_ref(job["supervisor_state"], metadata=True)
        exact_ref(job["gnu_complete_child_cost"], metadata=True)
        require(type(job.get("complete_supervised_invocation_monotonic_seconds")) in (int, float)
                and 0 <= job["complete_supervised_invocation_monotonic_seconds"] <= 950 + 30,
                "Supervised stage clock missing/out of scope")
        require(type(summary.get("elapsed_before_final_seal_seconds")) in (int, float)

                "Cooperative stage clock missing/out of scope")
        public = summary["details"].get("public_complete_return_seconds")

            require(type(public) in (int, float) and 0 <= public <= 900,
                    "Complete public-call return clock missing/out of scope")
        stages[name] = {"marker": small_ref(summary_path), "supervisor_state": job["supervisor_state"],
                        "gnu_complete_child_cost": job["gnu_complete_child_cost"],

                        "public_complete_return_seconds": public,

            and type(gate.get("bounded_100_draw_probe_permitted")) is bool,
            "Cost gate measurements/forecast scope differs")
    for name, entry in gate["measurement_receipts"].items():
        require(entry["complete_public_return_seconds"] == stages[name]["public_complete_return_seconds"],
                "Gate public-return cost differs from stage")
    require(marker["native_arithmetic_replay_verified"] is False
            and marker["native_likelihood_effects_attested"] is False and marker["interval"] is None,
            "Study overstates science")
    return {"final_marker": small_ref(path / "summary.json"), "stages": stages,
            "freeze": freeze, "freeze_reference": binding_ref, "cost_gate": gate_ref,
            "gate_values": gate}


def current_modules() -> list[str]:
    names = git("ls-files", "--", "test").decode().splitlines()
    # Match the runner's actual `rg --files test` inventory, including the
    # worktree rather than silently substituting a Git-only test list.
    rg_names = subprocess.check_output(["rg", "--files", "test"], cwd=ROOT).decode().splitlines()
    result = sorted(p for p in rg_names if Path(p).name.startswith("test_") and p.endswith(".py"))
    require(len(result) == len(set(result)) == 71 and set(result) <= set(names),
            "Current sorted rg 71-module inventory differs")
    return result

def static_checks(path: Path, digests: dict[str, str]) -> dict:
    report = load(path)
    rows = report.get("checks")
    require(report.get("schema") == "b3_common_source_static_v1"
            and report.get("all_passed") is True
            and isinstance(rows, list) and len(rows) == 3, "Final static report differs")
    commands = [row.get("command") for row in rows]
    ruff = {"ruff", "/home/joey/micromamba/bin/ruff"}
    suffixes = (("check",), ("format", "--check"),
                ("-m", "mypy", "--follow-imports=silent",
                 "--ignore-missing-imports", "--explicit-package-bases"))
    for number, (row, command, suffix) in enumerate(zip(rows, commands, suffixes, strict=True)):
        valid_executable = (command[0] in ruff if number < 2
                            else command[0] == ".venv/bin/python") if command else False
        require(type(row.get("return_code")) is int and row["return_code"] == 0
                and isinstance(command, list) and valid_executable
                and tuple(command[1:1 + len(suffix)]) == suffix
                and len(command[1 + len(suffix):]) == 5
                and tuple(command[1 + len(suffix):]) == STATIC_PATH_ORDER,
                "Static check command, paths or result differs")

        require(recorded.get(str((ROOT / name).resolve())) == digests[name],

            "commands": commands, "successful_integer_return_codes": [0, 0, 0]}

        finish_recheck()
        verify_owned_output()
        require(read_small(args.output) == output_data, "Exact owned evidence output bytes changed")
        finish_recheck()
        verify_owned_output()
        require(read_small(args.output) == output_data, "Owned evidence bytes changed at final return seal")
        verify_owned_output()
        os.close(directory_fd)
        directory_closed = True
    except BaseException:
        invalidate_owned_output()
        raise
    finally:
        if not directory_closed:
            try:
                os.close(directory_fd)
            except OSError:
                pass


if __name__ == "__main__":
    main()
