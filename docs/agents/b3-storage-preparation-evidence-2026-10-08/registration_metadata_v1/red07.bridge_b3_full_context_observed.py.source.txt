class RegistrationUnavailable(ValueError):
    """A metadata-only inspection refused missing or independently unadmitted facts."""

    def __init__(self, missing_gates, inspection):
        self.missing_gates = tuple(sorted(missing_gates))
        self.inspection = inspection
        super().__init__("Registration remains unavailable: " + ", ".join(self.missing_gates))


class _Budget:
    def __init__(self, parent: Path, max_seconds: float):
        if type(max_seconds) not in (int, float) or not math.isfinite(max_seconds) or not 0 < max_seconds <= 900:
            raise ValueError("Registration seconds must be positive and at most 900")
        self.deadline = time.monotonic() + max_seconds
        self.parent = parent
        self.check()

    def check(self):
        if time.monotonic() >= self.deadline:
            raise TimeoutError("Registration inspection reached its original deadline")
        if resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024 >= 4 * 1024**3:
            raise MemoryError("Registration inspection exceeds 4 GiB RSS")
        available = next(
            (int(line.split()[1]) * 1024 for line in Path("/proc/meminfo").read_text().splitlines() if line.startswith("MemAvailable:")),
            0,
        )
        if available < 4 * 1024**3:
            raise MemoryError("Registration inspection requires 4 GiB available host RAM")
        if shutil.disk_usage(self.parent).free - 2 * MIB < 20 * 1024**3:
            raise OSError("Registration inspection requires 20 GiB free disk after reservation")


def _path(value: str) -> Path:
    if type(value) is not str or not value or value.strip() != value or len(value.encode()) > 4096:
        raise ValueError("Require a bounded canonical absolute path")
    path = Path(value)
    if not path.is_absolute() or str(path) != value or ".." in path.parts or value.startswith("//"):
        raise ValueError("Require a canonical absolute path without aliases")
    if path.resolve() != path:
        raise ValueError("Canonical paths cannot use symlink aliases")
    return path


def _json(raw: bytes) -> dict:
    def unique(pairs):
        record = {}
        for key, value in pairs:
            if key in record:
                raise ValueError("Duplicate registration JSON key")
            record[key] = value
        return record

    def invalid(value):
        raise ValueError("Nonfinite registration JSON number: " + value)

    value = json.loads(raw, object_pairs_hook=unique, parse_constant=invalid)
    if type(value) is not dict:
        raise ValueError("Require a closed registration JSON object")
    return value


def _closed(value, fields, role):
    if type(value) is not dict or set(value) != fields:
        raise ValueError("Require the exact closed " + role + " fields")
    return value


def _sha(value, role):
    if type(value) is not str or re.fullmatch(r"[0-9a-f]{64}", value) is None:
        raise ValueError("Require a lowercase SHA256 for " + role)
    return value


def _ref(value, role):
    _closed(value, {"path", "sha256", "bytes"}, role + " Ref")
    path = _path(value["path"])
    _sha(value["sha256"], role)
    if type(value["bytes"]) is not int or not 0 <= value["bytes"] <= 64 * 1024**3:
        raise ValueError("Require a strict bounded nonnegative Ref size")
    try:
        current = path.lstat()
    except OSError as error:
        raise ValueError("Require an existing original " + role + " Ref") from error
    if not stat.S_ISREG(current.st_mode):
        raise ValueError("Require an original regular " + role + " Ref")
    if current.st_size != value["bytes"]:
        raise ValueError("Original " + role + " Ref size differs from stored bytes")
    return {"path": str(path), "sha256": value["sha256"], "bytes": value["bytes"]}


def _read_ref(inputs, value, role):
    reference = _ref(value, role)
    raw = inputs.read(Path(reference["path"]))
    if len(raw) != reference["bytes"]:
        raise ValueError("Original " + role + " Ref size differs from its admitted bytes")
    if sha256(raw).hexdigest() != reference["sha256"]:
        raise ValueError("Original " + role + " Ref hash differs")
    return raw


def _filemap(value, expected=None):
    if type(value) is not dict or not 1 <= len(value) <= 512 or any(type(key) is not str for key in value):
        raise ValueError("Require a bounded explicit consumer FileMap")
    if list(value) != sorted(value):
        raise ValueError("Consumer FileMap must be canonically sorted")
    if expected is not None and set(value) != {str(path) for path in expected}:
        raise ValueError("Consumer FileMap requires the exact 80 original source paths")
    for path, digest in value.items():
        _path(path)
        _sha(digest, "consumer FileMap")
    return value


def _family_helpers(raw):
    if sha256(raw).hexdigest() != FROZEN_FAMILY_SHA256:
        raise ValueError("Frozen genuine family metadata helper source differs")
    parsed = ast.parse(raw, filename=str(FAMILY_SOURCE))
    names = {"digest", "_name", "validate_family"}
    functions = [node for node in parsed.body if isinstance(node, ast.FunctionDef) and node.name in names]
    constants = [
        node for node in parsed.body
        if isinstance(node, ast.Assign) and len(node.targets) == 1
        and isinstance(node.targets[0], ast.Name) and node.targets[0].id == "MAX_COMPARISONS"
    ]
    if {node.name for node in functions} != names or len(functions) != 3 or len(constants) != 1:
        raise ValueError("Genuine family metadata projection is unavailable")
    projection = ast.Module(body=[*constants, *functions], type_ignores=[])
    namespace = {"json": json, "sha256": sha256, "Path": Path}
    exec(compile(projection, str(FAMILY_SOURCE), "exec"), namespace)
    return namespace


def _family(family, expected, helpers, inputs, output, profile):
    _sha(expected, "family digest")
    helpers["validate_family"](family, expected)
    if re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_.-]{0,127}", family["family_id"]) is None:
        raise ValueError("Require a canonical family ID")
    if profile == "synthetic_stored_arithmetic_v1" and family["model_arm"] != "base":
        raise ValueError("Synthetic registration permits only the base model arm")
    pairs, keys = set(), set()
    for comparison in family["comparisons"]:
        if re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_.-]{0,127}", comparison["comparison_id"]) is None:
            raise ValueError("Require a canonical comparison ID")
        left, right = (_path(comparison[name]) for name in ("bundle_a", "bundle_b"))
        pair = tuple(sorted((str(left), str(right))))
        if pair in pairs:
            raise ValueError("Duplicate or reversed original scientific comparison")
        pairs.add(pair)
        for key in (left, right):
            if output == key or output.is_relative_to(key) or key.is_relative_to(output):
                raise ValueError("Output overlaps an original planned source key")
            if os.path.lexists(key):
                raise ValueError("Original planned source key already exists; refusing retrospective registration")
            keys.add(str(key))
        for role, hash_key in (("table", "table_sha256"), ("paired_preflight", "paired_preflight_sha256")):
            path = _path(comparison[role])
            raw = inputs.read(path)
            if sha256(raw).hexdigest() != comparison[hash_key]:
                raise ValueError("Original family paired input hash differs")
    if len(keys) > 32:
        raise ValueError("Original source-key count exceeds its bound")
    return sorted(keys)


def _source_keys(value, keys, role):
    if type(value) is not dict or set(value) != set(keys) or list(value) != keys:
        raise ValueError("Require the exact sorted original family source-key closure for " + role)
    return value


def _refs(value, role):
    if type(value) is not list or not 1 <= len(value) <= 128:
        raise ValueError("Require a bounded nonempty " + role + " Ref closure")
    references = [_ref(item, role) for item in value]
    names = [item["path"] for item in references]
    if names != sorted(set(names)):
        raise ValueError("Require sorted unique " + role + " Ref paths")
    return references


def _records(value, fields, role, ref_fields):
    if type(value) is not list or not 1 <= len(value) <= 128:
        raise ValueError("Require a bounded nonempty " + role + " record list")
    for item in value:
        _closed(item, fields, role)
        for field in ref_fields:
            _ref(item[field], role + " " + field)

class _Inputs:
    def __init__(self, budget: _Budget):
        self.budget = budget
        self.files = {}
        self.retained_bytes = 0

    def read(self, path: Path, limit: int = METADATA_LIMIT) -> bytes:
        self.budget.check()
        path = _path(str(path))
        if path in self.files:
            self.seal()
            return self.files[path][4]
        if len(self.files) >= 128:
            raise MemoryError("Registration input file count exceeds its bound")
        before = path.lstat()
        if not stat.S_ISREG(before.st_mode) or before.st_size > limit:
            raise ValueError("Require a bounded original regular metadata file")
        if self.retained_bytes + before.st_size > 16 * MIB:
            raise MemoryError("Registration retained input bytes exceed their bound")
        fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
        identity = (before.st_dev, before.st_ino)
        self.files[path] = (fd, identity, before.st_size, None, None)
        opened = os.fstat(fd)
        if not stat.S_ISREG(opened.st_mode) or (opened.st_dev, opened.st_ino, opened.st_size) != (*identity, before.st_size):
            raise ValueError("Original metadata storage changed during admission")
        raw = bytearray()
        while block := os.read(fd, min(65536, limit + 1 - len(raw))):
            self.budget.check()
            if len(raw) + len(block) > limit:
                raise ValueError("Original metadata exceeds its bounded size")
            raw.extend(block)
        if len(raw) != before.st_size:
            raise ValueError("Original metadata bytes changed during admission")
        data = bytes(raw)
        self.files[path] = (fd, identity, before.st_size, sha256(data).hexdigest(), data)
        self.retained_bytes += len(data)
        return data

    def seal(self):
        for path, (fd, identity, size, digest, _) in self.files.items():
            self.budget.check()
            opened, current = os.fstat(fd), path.lstat()
            if (opened.st_dev, opened.st_ino, opened.st_size) != (*identity, size) or (current.st_dev, current.st_ino, current.st_size) != (*identity, size):
                raise ValueError("Original metadata pathname or descriptor changed")
            if _path(str(path)) != path:
                raise ValueError("Original metadata alias changed")
            if digest is not None:
                os.lseek(fd, 0, os.SEEK_SET)
                actual = sha256()
                count = 0
                while block := os.read(fd, 65536):
                    self.budget.check()
                    count += len(block)
                    if count > size:
                        raise ValueError("Original metadata bytes grew before refusal")
                    actual.update(block)
                if count != size or actual.hexdigest() != digest:
                    raise ValueError("Original metadata bytes changed before refusal")

    def close(self):
        failure = None
        for fd, identity, _, _, _ in self.files.values():
            try:
                current = os.fstat(fd)
                if not stat.S_ISREG(current.st_mode) or (current.st_dev, current.st_ino) != identity:
                    raise RuntimeError("Preserved foreign input descriptor during cleanup")
                os.close(fd)
            except BaseException as error:
                if failure is None:
                    failure = error
        if failure is not None:
            raise failure


def run(request_path: Path, output: Path, *, max_seconds=900):
    """Inspect one persistent request; never manufacture registration evidence."""
    output = _path(str(Path(output).absolute()))
    if not output.parent.is_dir():
        raise ValueError("Registration output parent must already exist")
    budget = _Budget(output.parent, max_seconds)
    inputs = _Inputs(budget)
    try:
        request = _json(inputs.read(Path(request_path).absolute()))
        _closed(request, REQUEST_FIELDS, "request")
        if request["schema"] != "b3_full_context_family_registration_request_v2":
            raise ValueError("Unsupported registration request schema")
        if request["phase"] not in ("register", "replay_registration"):
            raise ValueError("Unsupported registration phase; comparison requires admitted owner authority")
        if request["registration_profile"] not in PROFILES:
            raise ValueError("Unsupported registration profile")
        if request["phase"] == "register" and request["execution_catalog"] is not None:
            raise ValueError("Register cannot supply an execution catalog")
        family = _json(_read_ref(inputs, request["family"], "family"))
        consumers = _filemap(request["consumer_file_sha256"], SOFTWARE)
        for name, expected in consumers.items():
            raw = inputs.read(Path(name))
            if sha256(raw).hexdigest() != expected:
                raise ValueError("Original consumer source hash differs")
        helpers = _family_helpers(inputs.read(FAMILY_SOURCE))
        keys = _family(family, request["family_sha256"], helpers, inputs, output, request["registration_profile"])
        bindings = _source_keys(request["source_bindings"], keys, "source_bindings")
        intents = _source_keys(request["producer_intent"], keys, "producer_intent")
        _predecessors(request, keys)
        missing = {"genuine_preparation_reconciliation", "independent_source_admission", "registration_publication_and_replay"}
        for field, gate in (("preparation", "genuine_preparation"), ("issuer_source_admission", "accepted_issuer_source_admission"), ("decision_evidence", "decision_evidence"), ("rng_contract", "rng_contract")):
            if request[field] is None:
                missing.add(gate)
        if any(value is None for value in bindings.values()):
            missing.add("complete_source_bindings")
        if any(value is None for value in intents.values()):
            missing.add("complete_producer_intent")
        inspection = {
            "scope": "request_and_frozen_family_metadata_only",
            "registration_profile": request["registration_profile"],
