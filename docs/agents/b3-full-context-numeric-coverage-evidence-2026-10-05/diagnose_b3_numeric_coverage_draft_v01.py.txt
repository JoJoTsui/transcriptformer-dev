"""Small existing-library CPU diagnostic; complete numeric admission stays false."""

from hashlib import sha256
from importlib.metadata import version
import gc
import json
import os
from pathlib import Path
import shutil
import sys
import tempfile
import time
import tracemalloc

BASE = Path(__file__).resolve().parent
RESULT = BASE / "numeric_coverage_draft_v01_result.json"
NUMERIC_CAP = 200 * 1024**2


def main():
    started = time.monotonic()
    deadline = started + 900
    source_before = sha256(Path(__file__).read_bytes()).hexdigest()
    if RESULT.exists():
        raise FileExistsError(RESULT)
    if any(name in sys.modules for name in ("numpy", "torch", "h5py")):
        raise ValueError("Start this diagnostic before numeric packages are imported")
    if (
        any(
            os.environ.get(name) != "1"
            for name in ("OMP_NUM_THREADS", "OPENBLAS_NUM_THREADS", "MKL_NUM_THREADS", "NUMEXPR_NUM_THREADS")
        )
        or os.environ.get("CUDA_VISIBLE_DEVICES") != ""
    ):
        raise ValueError("Require one math thread and CUDA off before process startup")

    def check():
        if time.monotonic() >= deadline:
            raise TimeoutError("Original diagnostic 900-second deadline reached")
        available = next(
            int(row.split()[1]) * 1024
            for row in Path("/proc/meminfo").read_text().splitlines()
            if row.startswith("MemAvailable:")
        )
        rss = next(
            int(row.split()[1]) * 1024
            for row in Path("/proc/self/status").read_text().splitlines()
            if row.startswith("VmRSS:")
        )
        if available < 4 * 1024**3 or rss > 4 * 1024**3:
            raise MemoryError("Keep 4 GiB available RAM floor and diagnostic process RSS cap")
        if shutil.disk_usage(BASE).free < 20 * 1024**3:
            raise OSError("Keep 20 GiB free-disk floor")
        return {"observed_process_rss_bytes": rss, "observed_available_ram_bytes": available}

    check()

    def admit_explicit_request(size):
        if type(size) is not int or not 0 <= size <= NUMERIC_CAP:
            raise MemoryError("Explicit diagnostic buffer exceeds the unchanged 200 MiB boundary")
        check()

    boundary_refused = False
    try:
        admit_explicit_request(NUMERIC_CAP + 1)
    except MemoryError:
        boundary_refused = True
    if not boundary_refused:
        raise ValueError("Explicit oversized request must be refused without allocation")
    tracemalloc.start(1)
    tracing_before_numeric_imports = True
    import numpy as np
    import torch
    import h5py

    torch.set_num_threads(1)
    torch.set_num_interop_threads(1)
    check()
    domain = int(np.lib.tracemalloc_domain)

    def sample(label):
        resources = check()
        current, peak = tracemalloc.get_traced_memory()
        snapshot = tracemalloc.take_snapshot()
        domains = {}
        for trace in snapshot.traces:
            domains[trace.domain] = domains.get(trace.domain, 0) + trace.size
        del snapshot
        gc.collect()
        resources.update(
            {
                "label": label,
                "sampled_domain_live_bytes": {str(k): v for k, v in sorted(domains.items())},
                "traced_all_domain_current_before_snapshot_bytes": current,
                "traced_all_domain_peak_before_snapshot_bytes": peak,
                "tracer_metadata_current_bytes": tracemalloc.get_tracemalloc_memory(),
            }
        )
        return resources

    samples = [sample("after_numeric_imports")]
    observations = {}
    admit_explicit_request(4 * 1024**2)
    owner = np.ones(4 * 1024**2, dtype=np.uint8)
    alias = owner[::2]
    observations["numpy_owned_buffer"] = {
        "owner_bytes": int(owner.nbytes),
        "view_logical_bytes": int(alias.nbytes),
        "shares_memory": bool(np.shares_memory(owner, alias)),
        "view_owns_data": bool(alias.flags.owndata),
    }
    samples.append(sample("numpy_owner_and_alias_live"))
    admit_explicit_request(1024**2)
    external = bytearray(1024**2)
    external_view = np.frombuffer(external, dtype=np.uint8)
    observations["external_python_buffer"] = {
        "owner_bytes": len(external),
        "view_logical_bytes": int(external_view.nbytes),
        "view_owns_data": bool(external_view.flags.owndata),
        "view_base_is_memoryview": type(external_view.base).__name__ == "memoryview",
    }
    samples.append(sample("external_buffer_and_numpy_view_live"))
    admit_explicit_request(1024**2)
    tensor = torch.ones((256, 512), dtype=torch.float64, device="cpu")
    tensor_view = tensor.view(-1)
    storage = tensor.untyped_storage()
    view_storage = tensor_view.untyped_storage()
    observations["torch_cpu_storage"] = {
        "storage_bytes": int(storage.nbytes()),
        "view_storage_bytes": int(view_storage.nbytes()),
        "same_storage_pointer": storage.data_ptr() == view_storage.data_ptr(),
        "device": str(tensor.device),
    }
    samples.append(sample("torch_cpu_storage_and_alias_live"))
    with tempfile.TemporaryDirectory(prefix="b3-numeric-coverage-", dir=BASE) as temporary:
        temporary = Path(temporary)
        admit_explicit_request(1024**2)
        mapped = np.memmap(temporary / "mapping.bin", dtype=np.uint8, mode="w+", shape=(1024**2,))
        mapped[:] = 1
        mapped.flush()
        observations["numpy_memory_mapping"] = {
            "logical_bytes": int(mapped.nbytes),
            "owns_data": bool(mapped.flags.owndata),
            "backing_type": type(mapped.base).__name__,
            "mapping_file_bytes": (temporary / "mapping.bin").stat().st_size,
        }
        samples.append(sample("memory_mapping_live"))
        with h5py.File(temporary / "cache.h5", "w", rdcc_nbytes=65536, rdcc_nslots=521) as handle:
            for name in ("left", "right"):
                check()
                handle.create_dataset(name, shape=(8192,), dtype="float64", chunks=(512,), fillvalue=1)
        with h5py.File(temporary / "cache.h5", "r", rdcc_nbytes=65536, rdcc_nslots=521) as handle:
            left, right = handle["left"], handle["right"]
            capacities = []
            for dataset in (left, right):
                access = dataset.id.get_access_plist()
                try:
                    capacities.append(int(access.get_chunk_cache()[1]))
                finally:
                    access.close()
            admit_explicit_request(8192)
            left_data, right_data = left[:512], right[:512]
            observations["hdf5_two_datasets"] = {
                "configured_raw_chunk_cache_capacity_bytes_per_dataset": capacities,
                "configured_cache_capacity_sum_bytes": sum(capacities),
                "actual_cache_occupancy_bytes": None,
                "explicit_read_buffers_bytes": int(left_data.nbytes + right_data.nbytes),
            }
            samples.append(sample("two_hdf5_dataset_caches_and_read_buffers_live"))
            del left_data, right_data, left, right
        del mapped
        gc.collect()
    samples.append(sample("mapping_and_hdf5_closed"))
    admit_explicit_request(8 * 1024**2)
    tracemalloc.reset_peak()
    before_transient = tracemalloc.get_traced_memory()
    transient = np.ones(8 * 1024**2, dtype=np.uint8)
    while_transient = tracemalloc.get_traced_memory()
    samples.append(sample("explicit_transient_live"))
    del transient
    gc.collect()
    after_transient = tracemalloc.get_traced_memory()
    samples.append(sample("explicit_transient_freed"))
    observations["transient"] = {
        "explicit_buffer_bytes": 8 * 1024**2,
        "all_traced_before_current_bytes": before_transient[0],
        "all_traced_while_live_current_bytes": while_transient[0],
        "all_traced_while_live_peak_bytes": while_transient[1],
        "all_traced_after_free_current_bytes": after_transient[0],
        "all_traced_after_free_peak_bytes": after_transient[1],
        "peak_scope": "All tracemalloc domains since explicit reset; not a numeric or NumPy-domain peak",
    }
    del tensor_view, view_storage, storage, tensor, external_view, external, alias, owner
    gc.collect()
    samples.append(sample("explicit_python_numpy_torch_owners_released"))
    tracemalloc.stop()
    final_resources = check()
    source_after = sha256(Path(__file__).read_bytes()).hexdigest()
    if source_before != source_after:
        raise ValueError("Diagnostic source changed during actual execution")
    record = {
        "schema": "b3_numeric_allocator_coverage_diagnostic_v1",
        "status": "completed_diagnostic_only",
        "source_sha256_before": source_before,
        "source_sha256_after": source_after,
        "package_versions": {n: version(n) for n in ("numpy", "torch", "h5py")},
        "tracing_started_before_numeric_imports": tracing_before_numeric_imports,
        "numpy_tracemalloc_domain": domain,
        "samples": samples,
        "observations": observations,
        "explicit_200mib_plus_one_request_refused_without_allocation": boundary_refused,
        "known_explicit_simultaneous_buffer_upper_bound_bytes": 16 * 1024**2 + 8192,
        "complete_live_numeric_peak_bytes": None,
        "complete_numeric_allocation_chronology": False,
        "complete_native_allocation_guard": False,
        "complete_numeric_census_granted": False,
        "source_admission_granted": False,
        "runtime_admission_granted": False,
        "checkpoint_tensors_loaded": False,
        "model_forwards_performed": False,
        "project_data_loaded": False,
        "explicit_gpu_device_queries_performed": False,
        "unknown_coverage": [
            "native allocator scratch/caches and library import state",
            "HDF5 actual cache occupancy and native allocations",
            "PyTorch CPU storage allocation/deallocation chronology",
            "memory mappings not covered by array data-domain tracing",
            "transient allocations between snapshots",
            "complete parent/producer/helper process scope",
        ],
        "diagnostic_body_seconds_before_result_io_and_final_cleanup": time.monotonic() - started,
        "final_resource_sample": final_resources,
        "scope": "Small isolated existing-library CPU diagnostic; samples and explicit storage arithmetic are not whole-operation census/admission",
    }
    with RESULT.open("x") as handle:
        json.dump(record, handle, indent=2, allow_nan=False)
        handle.write("\n")
    print(
        json.dumps(
            {
                "status": record["status"],
                "complete_numeric_census_granted": False,
                "samples": len(samples),
                "result": str(RESULT),
            }
        )
    )


if __name__ == "__main__":
    main()
