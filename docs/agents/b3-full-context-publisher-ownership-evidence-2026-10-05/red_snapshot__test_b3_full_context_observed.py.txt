def _canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()


def _ref(path):
    path = Path(path).resolve()
    data = path.read_bytes()
    return {"path": str(path), "sha256": sha256(data).hexdigest(), "bytes": len(data)}


def _write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(_canonical(value) + b"\n")
    return _ref(path)


def _module():
    # Explicitly fail at the public file seam before any numeric work if the
    # reviewed draft has not been installed. There is no fabricated file ref.
    assert PUBLIC_SOURCE.is_file(), "Review/copy the draft to its canonical public source first"
    from scripts import publish_b3_full_context_observed

    return publish_b3_full_context_observed


def _request(native_path, folder, *, ranges=((0, 1), (1, 4)), phase="publish", execution=None):
    path = make_common_source_request(
        native_path,
        list(ranges),
        folder,
        phase="replay" if phase == "replay" else "build",
        execution_catalog=execution,
    )
    body = json.loads(path.read_bytes())
    public = _module()
    body.update(schema=public.SCHEMA, phase=phase, execution_catalog=execution)
    body["consumer_file_sha256"][str(PUBLIC_SOURCE)] = _ref(PUBLIC_SOURCE)["sha256"]
    assert set(body["consumer_file_sha256"]) == {str(path) for path in public.SOFTWARE}
    assert len(body["consumer_file_sha256"]) == 70
    _write(path, body)
    return path


def _all_rows(summary):
    root = json.loads(Path(summary["rows"]["path"]).read_bytes())
    return [
        row for descriptor in root["pages"] for row in json.loads(Path(descriptor["file"]["path"]).read_bytes())["rows"]
    ]


def _assert_oracle_rows(rows, oracle):
    for actual, expected in zip(rows, oracle["gene_results"], strict=True):
        for key in (
            "gene_id",
            "focal_scored_cells",
            "focal_scored_embryos",
            "candidate_peers",
            "matched_peers",
            "positive_contrast_peers",
            "unavailable_reason",
        ):
            assert actual[key] == expected[key]
        for key in ("raw_impact_bits", "null_mean_impact_bits", "null_sample_sd_bits", "diagnostic_z"):
            reference_key = "null_corrected_z" if key == "diagnostic_z" else key
            value = expected[reference_key]
            assert (actual[key] is None) == (value is None)
            if value is not None:
                assert struct.pack("<d", actual[key]) == struct.pack("<d", value)


def _assert_no_complete_output(output):
    assert not (output / "summary.json").exists()
    assert not output.with_name(output.name + ".claim").exists()
    assert not list(output.parent.glob(".b3-full-context-observed-*"))


def _assert_no_owned_modules(before):
    assert not [
        name
        for name in sys.modules
        if name.startswith(("_b3_authenticated_", "_b3_observed_helpers_")) and name not in before
    ]
    for name, value in before.items():
        if name == "scripts" or name.startswith(("scripts.", "transcriptformer")):
            assert sys.modules.get(name) is value


def _finite_native_fixture(folder, *, unsafe_gene=None):

    The baseline supplies portable source/plan metadata. This new original
    fixture replaces every affected gene axis, metric, support bit, attempt,
    proof, native counter and actual file binding before a new native catalog
    is declared. No kernel, digest or admission checker is patched. Scalar
    impacts are stored test input; their checkpoint effects are unattested.
    """
    import h5py
    import numpy as np
    from scripts.b3_native_catalog_pages import run as declare_catalog
    from test.test_b3_paged_native_cache import RECORD_DTYPE

    native_path, native, entries, full = fixture(folder, n_cells=64)
    root = json.loads(Path(native["catalog"]["path"]).read_bytes())
    context_path = Path(native["context_request"]["path"])
    context = json.loads(context_path.read_bytes())
    plan_path = Path(root["plan"]["path"])
    plan = json.loads(plan_path.read_bytes())
    config_path, full_path, paired_path = [
        Path(plan[key + "_path"]) for key in ("config", "full_preflight", "paired_preflight")
    ]
    config = json.loads(config_path.read_bytes())
    paired = json.loads(paired_path.read_bytes())
    genes, embryos, cells = [f"ENSG{i:011d}" for i in range(1, 65)], ["emb0", "emb1"], 64
    if unsafe_gene is not None:
        # Internal separators are accepted by the frozen native identifier
        # admission. Keep that actual original axis and bind every affected
        # metric/proof/support/source record; only the new TSV seam refuses it.
        genes[0] = unsafe_gene
    raw_genes, scored_genes = 63, 62
    method = plan["method"]
    config["gene_ids"] = genes
    config_ref = _write(config_path, config)
    full["cohort_contract"]["gene_ids_sha256"] = sha256(_canonical(genes)).hexdigest()
    cohort_sha = sha256(_canonical(full["cohort_contract"])).hexdigest()
    cell_embryo = np.arange(cells, dtype="<i4") % 2
    counts = np.bincount(cell_embryo, minlength=2).astype("<i8")
    raw = np.zeros((len(genes), cells), dtype="u1")
    raw[:raw_genes] = 1
    support = np.zeros_like(raw)
    support[:scored_genes] = 1
    support_path = Path(plan["support_h5_path"])
    with h5py.File(support_path, "w") as handle:
        handle.attrs.update(
            schema="b3_measured_zero_full_support_v1",
            method=method,
            bitorder="little",
            cohort_sha256=cohort_sha,
            cell_order="sorted source/prepared paths then surviving phase rows in native row order",

        handle.create_dataset("embryo_ids", data=np.asarray(embryos, dtype=h5py.string_dtype()))
        handle.create_dataset("raw_positive", data=np.packbits(raw, axis=1, bitorder="little"))
        handle.create_dataset("native_scorable_support", data=np.packbits(support, axis=1, bitorder="little"))
        handle.create_dataset("cell_embryo_index", data=cell_embryo)
        handle.create_dataset("cell_source_index", data=np.zeros(cells, dtype="<i4"))
        handle.create_dataset("cell_source_row_index", data=np.arange(cells, dtype="<i8"))
    support_ref = _ref(support_path)
    full.update(
        config_sha256=config_ref["sha256"],
        cohort_sha256=cohort_sha,
        n_frozen_genes=len(genes),
        native_sequence_length=len(genes),
        estimated_raw_rows=raw_genes * cells,
    )
    full["support_h5"].update(sha256=support_ref["sha256"], shape=[len(genes), cells // 8])
    full["metrics"] = [
        {
            "gene_id": gene,
            "dropout": 0.0 if index < raw_genes else 1.0,
            "mean_log1p_normalized_expression": 1.0 if index < raw_genes else 0.0,
        }
        for index, gene in enumerate(genes)
    ]
    full["gene_support"] = [
        {
            "gene_id": gene,
            "raw_token_attempts": cells if index < raw_genes else 0,
            "raw_positive_cells": cells if index < raw_genes else 0,
            "potentially_scorable_cells": cells if index < scored_genes else 0,
            "potentially_scorable_embryos": len(embryos) if index < scored_genes else 0,
        }
        for index, gene in enumerate(genes)
    ]
    full_ref = _write(full_path, full)
    paired["inputs"] = {str(config_path): config_ref["sha256"], str(full_path): full_ref["sha256"]}
    paired["cohort_sha256"][0] = cohort_sha
    paired["prospective_statistic"]["genes_a"] = genes
    paired_ref = _write(paired_path, paired)
    plan.update(
        config_sha256=config_ref["sha256"],
        full_preflight_sha256=full_ref["sha256"],
        paired_preflight_sha256=paired_ref["sha256"],
        support_h5_sha256=support_ref["sha256"],
        cohort_sha256=cohort_sha,
        n_frozen_genes=len(genes),
        native_sequence_length=len(genes),
        estimated_raw_rows=raw_genes * cells,
        native_scorable_contrasts=scored_genes * cells,
    )
    plan["ranges"] = [
        {
            "index": cell,
            "start": cell,
            "stop": cell + 1,
            "max_positive_attempts": len(genes),
            "native_scorable_contrasts": scored_genes,
        }
        for cell in range(cells)
    ]
    plan_ref = _write(plan_path, plan)
    context["plans"][0] = {key: plan_ref[key] for key in ("path", "sha256")}
    context["input_file_sha256"] = {path: _ref(Path(path))["sha256"] for path in context["input_file_sha256"]}
    native["context_request"] = _write(context_path, context)
    provenance_path = Path(root["producer_provenance"]["path"])
    provenance = json.loads(provenance_path.read_bytes())
    provenance.update(plan_sha256=plan_ref["sha256"], config_sha256=config_ref["sha256"])
    provenance_ref = _write(provenance_path, provenance)
    provenance_sha = sha256(_canonical(provenance)).hexdigest()
    common = sorted([_ref(Path(ref["path"])) for ref in root["common_files"]], key=lambda ref: ref["path"])
    likelihood = np.asarray([-float(index + 1) for index in range(raw_genes)], dtype="<f8").tobytes()
    raw_bits = np.packbits(raw[:, 0], bitorder="little").tobytes().hex()
    zero_bits = np.packbits(~raw[:, 0].astype(bool), bitorder="little").tobytes().hex()
    for cell, entry in enumerate(entries):
        files = entry["files"]
        bounds = plan["ranges"][cell]
        header_path = Path(files["header.json"]["path"])
        header = json.loads(header_path.read_bytes())
        header.update(range=bounds, plan_sha256=plan_ref["sha256"])
        files["header.json"] = _write(header_path, header)
        records = np.asarray(

        proof_path = Path(files["proofs.jsonl"]["path"])
        proof = json.loads(proof_path.read_bytes())
        proof.update(
            producer_provenance_sha256=provenance_sha,
            raw_positive_bits=raw_bits,
            eligible_target_count=raw_genes,
            original_target_log_probs=likelihood.hex(),

            _canonical({"cell": cell, "genes": genes[:raw_genes], "stored_targets": likelihood.hex()})
        ).hexdigest()
        proof["raw_nonzero_row_sha256"] = sha256(
            _canonical({"n_features": len(genes), "nonzero": [[gene, 1.0] for gene in genes[:raw_genes]]})
        ).hexdigest()
        proof["attempt_target_id_hashes_sha256"] = sha256(
            _canonical([genes[:raw_genes] for _ in range(raw_genes)])
        ).hexdigest()
        files["proofs.jsonl"] = _write(proof_path, proof)
        footer_path = Path(files["footer.json"]["path"])
        footer = json.loads(footer_path.read_bytes())
        footer.update(
            record_count=raw_genes,
            header_sha256=files["header.json"]["sha256"],
            records_sha256=files["records.bin"]["sha256"],
            proofs_sha256=files["proofs.jsonl"]["sha256"],
        )
        files["footer.json"] = _write(footer_path, footer)
        cert_path = Path(files["certificate"]["path"])
        certificate = json.loads(cert_path.read_bytes())
        certificate.update(
            range=bounds,
            plan_sha256=plan_ref["sha256"],
            producer_provenance_sha256=provenance_sha,
            verified_input_file_sha256={
                ref["path"]: ref["sha256"]
                for ref in [
                    *common,
                    *[files[name] for name in ("header.json", "records.bin", "proofs.jsonl", "footer.json")],
                ]
            },
        )
        certificate["cells"][0].update(
            native_attempts=raw_genes, eligible_target_count=raw_genes, source_native_raw_zero_eligible_bits=zero_bits
        )
        files["certificate"] = _write(cert_path, certificate)
    manifest = folder / "finite-entries.jsonl"
    manifest.write_bytes(b"".join(_canonical(entry) + b"\n" for entry in entries))
    catalog_request = json.loads((folder / "catalog-request.json").read_bytes())

        plan=plan_ref, producer_provenance=provenance_ref, common_files=common, entries_manifest=_ref(manifest)
    )
    catalog_request_path = folder / "finite-catalog-request.json"
    _write(catalog_request_path, catalog_request)
    native["catalog"] = declare_catalog(catalog_request_path, folder / "finite-catalog")["catalog"]
    offsets = [min(gene, scored_genes) * cells for gene in range(len(genes) + 1)]
    arrays = {
        "gene_offsets.u64": np.asarray(offsets, dtype="<u8"),
        "cell_index.u32": np.tile(np.arange(cells, dtype="<u4"), scored_genes),
        "impact_bits.f64": np.repeat(np.arange(1, scored_genes + 1, dtype="<f8"), cells),
    }
    for name, array in arrays.items():
        path = Path(native["csr_arrays"][name]["path"])
        path.write_bytes(array.tobytes())
        native["csr_arrays"][name] = _ref(path)
    metric_path = Path(native["embryo_metrics_metadata"]["path"])
    metric = json.loads(metric_path.read_bytes())
    with h5py.File(metric_path.parent / "metrics.h5", "w") as handle:
        handle.attrs.update(
            schema=metric["schema"],

        handle.create_dataset("gene_ids", data=np.asarray(genes, dtype=h5py.string_dtype()))
        handle.create_dataset("embryo_ids", data=np.asarray(embryos, dtype=h5py.string_dtype()))
        handle.create_dataset("embryo_cell_counts", data=counts)
        expression, detected = np.zeros((2, len(genes)), dtype="<f8"), np.zeros((2, len(genes)), dtype="<i8")
        expression[:, :raw_genes] = counts[:, None]

        handle.create_dataset("expression_sum", data=expression)
        handle.create_dataset("detected", data=detected)
    metric.update(
        plan_sha256=plan_ref["sha256"],
        cohort_sha256=cohort_sha,
        n_frozen_genes=len(genes),
        gene_order_sha256=sha256(_canonical(genes)).hexdigest(),
        metrics_h5_sha256=_ref(metric_path.parent / "metrics.h5")["sha256"],

    }
    native["embryo_metrics_metadata"] = _write(metric_path, metric)
    _write(native_path, native)
    return SimpleNamespace(path=native_path, native=native, entries=entries, full=full, genes=genes, embryos=embryos)

def test_real_64_gene_finite_native_publication_matches_public_oracle_tsv_and_fresh_replay(tmp_path):
    from transcriptformer.finetune.b3_measured_zero_scores import score_bounded_measured_zero

    stored = _finite_native_fixture(tmp_path / "finite-native")
    public = _module()
    ranges = [(start, start + 8) for start in range(0, len(stored.genes), 8)]
    request = _request(stored.path, tmp_path / "publish-request", ranges=ranges)
    output = tmp_path / "published"
    before = dict(sys.modules)
    result = public.run(request, output)
    _assert_no_owned_modules(before)
    reference = _public_reference({"native_request": _ref(stored.path)})
    oracle = score_bounded_measured_zero(
        positive_rows=reference["rows"],
        cell_proofs=reference["proofs"],
        metrics=stored.full["metrics"],
        gene_ids=reference["gene_ids"],
    )
    rows = _all_rows(result)
    _assert_oracle_rows(rows, oracle)
    finite = [row for row in rows if row["diagnostic_z"] is not None]
    unavailable = [row for row in rows if row["diagnostic_z"] is None]
    assert len(rows) == len(stored.genes) >= 50
    assert len(finite) >= 2 and unavailable
    assert all(row["candidate_peers"] >= 50 and row["matched_peers"] >= 2 for row in finite)
    assert all(row["null_sample_sd_bits"] > 0 for row in finite)
    assert len({row["raw_impact_bits"] for row in finite}) >= 3
    state = json.loads(Path(result["unit_state"]["path"]).read_bytes())
    assert state["metrics"] == stored.full["metrics"] and state["bins"] == oracle["bins"]
    assert sum(row["status"] == "available" for row in state["bins"]) >= 50
    assert result["n_finite_scores"] == len(finite) == oracle["finite_null_scores"]
    table = Path(result["finite_scores"]["path"]).read_text().splitlines()
    assert table[0] == "gene_id\tnull_corrected_z" and len(table) == len(finite) + 1
    for line, expected in zip(table[1:], finite, strict=True):
        gene, value = line.split("\t")
        assert gene == expected["gene_id"]
        assert struct.pack("<d", float(value)) == struct.pack("<d", expected["diagnostic_z"])
    replay_request = _request(
        stored.path, tmp_path / "replay-request", ranges=ranges, phase="replay", execution=_ref(output / "summary.json")
    )
    replay_output = tmp_path / "replayed"
    replay = public.run(replay_request, replay_output)
    assert replay["validation"]["rows_compared"] == len(stored.genes)
    assert replay["flags"]["independent_observed_arithmetic_replay_verified"] is True
    assert _all_rows(replay) == rows
    for key in ("source", "unit_state", "finite_scores"):
        assert Path(replay[key]["path"]).read_bytes() == Path(result[key]["path"]).read_bytes()
    assert json.loads(replay_request.read_bytes())["focal_catalog"] == json.loads(request.read_bytes())["focal_catalog"]
    for receipt in (result, replay):
        assert receipt["flags"]["native_likelihood_effects_attested"] is False
        assert receipt["flags"]["native_arithmetic_replay_verified"] is False
        assert receipt["flags"]["fixed_observed_family_verified"] is False
        assert receipt["flags"]["scientific_readiness"] == "unavailable"
    assert len(stored.genes) < 500 and len(stored.embryos) < 5
    _assert_no_owned_modules(before)

@pytest.mark.parametrize("replacement", ["fifo", "symlink", "identical_regular"])
def test_descriptor_admission_refuses_storage_replacement_before_read(tmp_path, monkeypatch, replacement):
    public = _module()
    source, alternate = tmp_path / "source.json", tmp_path / "alternate.json"
    source.write_bytes(b"stored bytes")
    alternate.write_bytes(source.read_bytes())
    budget = public._Budget(tmp_path, 900, time.monotonic())
    original_open = os.open
    seen = []

    def replace_on_open(path, flags, *args, **kwargs):
        if Path(path) == source and not seen:
            # A regression must fail here before it can block on the FIFO.
            assert flags & os.O_NOFOLLOW and flags & os.O_NONBLOCK
            seen.append(flags)
            source.unlink()
            if replacement == "fifo":
                os.mkfifo(source)
            elif replacement == "symlink":
                source.symlink_to(alternate)
            else:
                os.replace(alternate, source)
        return original_open(path, flags, *args, **kwargs)

    monkeypatch.setattr(os, "open", replace_on_open)
    with pytest.raises((ValueError, OSError)):
        budget.actual(source, MIB)
    assert len(seen) == 1
    assert seen[0] & os.O_NOFOLLOW and seen[0] & os.O_NONBLOCK
    assert not budget.pins

def test_unit_metric_tolerance_cannot_silently_change_original_bin_identity(stored_native, tmp_path, monkeypatch):
    public = _module()
    original = public._bootstrap

    def alter_rebuilt_bin(request, budget):
        modules, consumers, old = original(request, budget)
        engine = modules["engine"]
        build = engine.build_expression_dropout_bins
        calls = []

        def altered(metrics):
            bins = build(metrics)
            calls.append(True)
            if len(calls) == 2:
                # Keep metrics identical; alter only the second reconstructed
                # bin disposition to exercise the real unit-bin admission seam.

            return bins

        engine.build_expression_dropout_bins = altered
        return modules, consumers, old

@pytest.mark.parametrize("phase", ["publish", "replay"])
def test_public_snapshot_pin_reservation_distinguishes_originals_copies_and_generated_paths(
    stored_native,
    completed,
    tmp_path,
    monkeypatch,
    phase,
):
    public = _module()
    reserve, bootstrap = public._reserve_pin_paths, public._bootstrap
    reservations, snapshots = [], []

    def observe_reservation(source, snapshot_request, workspace, staging, genes, budget):
        admitted = set(budget.pins)
        upper, original_refs = reserve(source, snapshot_request, workspace, staging, genes, budget)
        originals = {Path(ref["path"]) for ref in original_refs}
        csr = {Path(ref["path"]) for ref in snapshot_request["csr_arrays"].values()}
        manifest = {Path(source["source_commitment"]["entries_manifest"]["path"])}
        native_originals = csr | {
            Path(source["source_commitment"][name]["path"]) for name in ("support", "embryo_metrics_h5")
        }
        copies = {
            workspace / "snapshot" / name
            for name in ("gene_offsets.u64", "cell_index.u32", "impact_bits.f64", "support.h5", "metrics.h5")
        }
        generated = {
            staging / name for name in ("source.json", "unit-state.json", "scores.tsv", "rows.json", "summary.json")
        }
        generated.update(
            staging / f"row-page-{index:06d}.json"
            for index in range((genes + public.PAGE_SIZE - 1) // public.PAGE_SIZE)
        )
        assert len(native_originals) == len(copies) == 5
        assert native_originals | manifest <= originals
        assert len(generated) == 5 + (genes + public.PAGE_SIZE - 1) // public.PAGE_SIZE
        assert originals - admitted == csr | manifest and not ((csr | manifest) & admitted)
        expected = admitted | originals | copies | generated
        assert budget.pin_paths == expected and budget.pin_limit == len(expected)
        assert upper == len(expected) * 24
        assert budget.pin_paths - admitted == csr | manifest | copies | generated
        reservations.append((budget, originals))
        return upper, original_refs

    def observe_bootstrap(request, budget):
        modules, consumers, old = bootstrap(request, budget)
        enter = modules["producer"]._snapshot

        @contextmanager
        def observe_snapshot(*args, **kwargs):
            # The actual new originals are authenticated before payload reads,
            # while their private copies remain different planned pathnames.
            assert len(reservations) == 1 and reservations[0][0] is budget
            assert reservations[0][1].issubset(budget.pins)
            snapshots.append(True)
            with enter(*args, **kwargs) as native:
                yield native
                # Keep the real copy/kernel/hash/admission calls intact. Every
                # reserved identity is used once, with no count-only spare slot.
                assert set(budget.pins) == budget.pin_paths
                assert len(budget.pins) == budget.pin_limit

        modules["producer"]._snapshot = observe_snapshot
        return modules, consumers, old

    monkeypatch.setattr(public, "_reserve_pin_paths", observe_reservation)
    monkeypatch.setattr(public, "_bootstrap", observe_bootstrap)
    execution = _ref(completed.output / "summary.json") if phase == "replay" else None
    request = _request(stored_native.path, tmp_path / "request", phase=phase, execution=execution)
    before = dict(sys.modules)
    result = public.run(request, tmp_path / "output")
    assert snapshots == [True] and len(reservations) == 1
    assert 0 < result["numeric_working_upper_bytes"] <= 200 * MIB
    assert result["validation"]["rows_compared"] == (4 if phase == "replay" else 0)
    _assert_no_owned_modules(before)


def test_previous_range_arrays_are_released_before_next_range_and_failure_cleans_up(
    stored_native, tmp_path, monkeypatch
):
    public = _module()
    original = public._bootstrap
    first_arrays, calls, mappings = [], [], []
    before = dict(sys.modules)

    def observed_kernel(request, budget):
        modules, consumers, old = original(request, budget)
        engine = modules["engine"]
        build = engine._build_statistics
        snapshot = modules["producer"]._snapshot

        @contextmanager
        def capture_mappings(*args, **kwargs):
            with snapshot(*args, **kwargs) as native:
                mappings.extend(native["arrays"])
                assert all(not mapping._mmap.closed for mapping in mappings)
                yield native

        modules["producer"]._snapshot = capture_mappings

        def fail_second(*args, **kwargs):
            calls.append(True)
            if len(calls) == 2:
                assert all(reference() is None for reference in first_arrays)
                raise MemoryError("owned second-range allocation refusal")
            arrays = build(*args, **kwargs)
            first_arrays.extend(weakref.ref(array) for array in arrays.values())
            return arrays

        engine._build_statistics = fail_second
        return modules, consumers, old

    monkeypatch.setattr(public, "_bootstrap", observed_kernel)
    request = _request(stored_native.path, tmp_path / "request")
    output = tmp_path / "output"
    with pytest.raises(MemoryError, match="second-range"):
        public.run(request, output)
    assert len(calls) == 2
    assert len(mappings) == 3 and all(mapping._mmap.closed for mapping in mappings)
    _assert_no_complete_output(output)
    _assert_no_owned_modules(before)

def test_reused_private_child_descriptor_refusal_closes_every_other_known_owned_handle(
    stored_native, tmp_path, monkeypatch
):
    public = _module()
    request = _request(stored_native.path, tmp_path / "request")
    output = tmp_path / "output"
    seal = public._Publication.seal_published
    ordinary_fstat, ordinary_close, ordinary_open, ordinary_dup2 = os.fstat, os.close, os.open, os.dup2
    before = dict(sys.modules)
    owned, reused, snapshots = [], [], []
    foreign = tmp_path / "foreign-reused-child-fd.bin"
    foreign_bytes = b"caller-owned replacement descriptor must survive\n"
    foreign.write_bytes(foreign_bytes)

    def replace_actual_context_descriptor_before_private_cleanup(owner):
        seal(owner)
        if not reused:
            assert (output / "summary.json").is_file()
            assert owner.workspace is not None
            snapshot = owner.workspace / "snapshot"
            copied = list(snapshot.iterdir())
            assert len(copied) == 5 and all(path.is_file() for path in copied)
            snapshots.append((snapshot, copied))
            for name in (
                "workspace_publication_fd",
                "workspace_context_fd",
                "workspace_snapshot_fd",
                "workspace_fd",
                "parent_fd",
                "claim_fd",
                "output_fd",
                "staging_fd",
                "marker_fd",
            ):
                fd = getattr(owner, name)
                assert fd is not None
                opened = ordinary_fstat(fd)
                owned.append((name, fd, (opened.st_dev, opened.st_ino)))
            fd = owner.workspace_context_fd
            ordinary_close(fd)
            replacement = ordinary_open(foreign, os.O_RDONLY | os.O_NOFOLLOW)
            if replacement != fd:
                ordinary_dup2(replacement, fd)
                ordinary_close(replacement)
            opened = ordinary_fstat(fd)
            reused.append((fd, (opened.st_dev, opened.st_ino)))

    monkeypatch.setattr(public._Publication, "seal_published", replace_actual_context_descriptor_before_private_cleanup)
    try:
        with pytest.raises(RuntimeError, match="descriptor|identity|workspace"):
            public.run(request, output)
        assert len(owned) == 9 and len(reused) == 1
        foreign_fd, foreign_identity = reused[0]
        current = ordinary_fstat(foreign_fd)
        assert (current.st_dev, current.st_ino) == foreign_identity
        assert os.read(foreign_fd, len(foreign_bytes)) == foreign_bytes
        assert foreign.read_bytes() == foreign_bytes
        assert not (output / "summary.json").exists()
        assert not output.with_name(output.name + ".claim").exists()
        assert len(snapshots) == 1
        snapshot, copied = snapshots[0]
        # Context ownership is uncertain after its FD was reused. The original
        # snapshot remains independently bound and must be drained before
        # refusing, even while that context and its enclosing root are kept.
        assert not snapshot.exists()
        assert all(not path.exists() for path in copied)
        for name, fd, _identity in owned:
            if name != "workspace_context_fd":
                with pytest.raises(OSError) as caught:
                    ordinary_fstat(fd)
                assert caught.value.errno == errno.EBADF
        _assert_no_owned_modules(before)
    finally:
        # A failing RED implementation may leak the independently known
        # owned handles. Release only exact originals, never a reused foreign FD.
        for name, fd, identity in owned:
            if name == "workspace_context_fd":
                continue
            try:
                current = ordinary_fstat(fd)
            except OSError as error:
                assert error.errno == errno.EBADF
            else:
                if (current.st_dev, current.st_ino) == identity:
                    ordinary_close(fd)
        for fd, identity in reused:
            current = ordinary_fstat(fd)
            assert (current.st_dev, current.st_ino) == identity
            ordinary_close(fd)

    def fail_later_probe(fd):
        if acquired and fd == acquired[0][0] and not injected:
            assert birth_seen == [True]
            owner = acquired[0][2]
            assert owner.descriptor_identities["workspace_" + child_name + "_fd"] == acquired[0][1]
            injected.append(True)
            raise OSError(errno.EIO, "recoverable child birth followed by admission probe refusal")

    def observe_stat(path, *args, **kwargs):
        if acquired and type(path) is int and path == acquired[0][0] and not birth_seen:
            result = ordinary_stat(path, *args, **kwargs)
            birth_seen.append(True)
            return result
        if probe == "proc" and acquired and path == f"/proc/self/fd/{acquired[0][0]}":
            fail_later_probe(acquired[0][0])
        return ordinary_stat(path, *args, **kwargs)

    def observe_fstat(fd):
        if probe == "fstat":
            fail_later_probe(fd)
        return ordinary_fstat(fd)

    monkeypatch.setattr(public._Publication, "bind_workspace_child", observe_actual_binding)
    monkeypatch.setattr(os, "open", record_original_open)
    monkeypatch.setattr(os, "stat", observe_stat)
    monkeypatch.setattr(os, "fstat", observe_fstat)
    try:
        with pytest.raises(OSError, match="recoverable child birth followed by admission probe refusal"):
            public.run(request, output)
        assert len(acquired) == 1 and birth_seen == injected == [True]
        with pytest.raises(OSError) as caught:
            ordinary_fstat(acquired[0][0])
        assert caught.value.errno == errno.EBADF
        _assert_no_complete_output(output)
        _assert_no_owned_modules(before)
    finally:
        for fd, identity, _owner in acquired:
            try:
                current = ordinary_fstat(fd)
            except OSError as error:
                assert error.errno == errno.EBADF
            else:
                if (current.st_dev, current.st_ino) == identity:
                    ordinary_close(fd)

def _observe_authenticated_class(monkeypatch, public, class_name, install):
    """Observe actual same-buffer helper execution, without changing its source.

    Fault injection runs after the real authenticated body has executed. The
    frozen kernel, hashes, imports and numeric/source admission remain active.
    Nested registries execute that same body through the ordinary exec seam.
    """
    ordinary_exec = builtins.exec
    observed = set()

    def after_authenticated_exec(code, globals=None, locals=None, **kwargs):
        if globals is None:
            caller = sys._getframe(1)
            globals = caller.f_globals
            if locals is None:
                locals = caller.f_locals
        result = ordinary_exec(code, globals, locals, **kwargs)
        if globals.get("__file__") == str(public.HELPER_SOURCE):
            cls = globals.get(class_name)
            if cls is not None and id(cls) not in observed:
                observed.add(id(cls))
                install(cls)
        return result

    monkeypatch.setattr(builtins, "exec", after_authenticated_exec)
    return observed


@pytest.mark.parametrize("registration", ["source", "clone"])
def test_public_authenticated_child_registration_refuses_occupied_foreign_key(
    stored_native, tmp_path, monkeypatch, registration
):
    public = _module()
    request = _request(stored_native.path, tmp_path / "request")
    output, collisions = tmp_path / "output", []
    before = dict(sys.modules)

    def reserve(name):
        assert name not in sys.modules
        foreign = ModuleType("foreign-child-registration")
        sys.modules[name] = foreign
        collisions.append((name, foreign))
        return foreign

    if registration == "source":

        def install(cls):
            allocate = cls._allocate

            def occupied_source(registry, identity):
                foreign = None
                if not collisions and identity.startswith("scripts."):
                    name = f"_b3_authenticated_{id(registry)}_{identity.replace('.', '_')}"
                    foreign = reserve(name)
                result = allocate(registry, identity)
                if foreign is not None:
                    # v03 overwrites here; fail immediately so the RED run
                    # cannot hide the collision behind a later cleanup error.
                    assert sys.modules.get(name) is foreign
                return result

            cls._allocate = occupied_source

        observed = _observe_authenticated_class(monkeypatch, public, "AuthenticatedHelpers", install)
    else:

        def install(cls):
            register = cls.__setitem__

            def occupied_clone(cache, key, value):
                registry = cache.registry
                needs_clone = (
                    value.__name__ not in registry.module_names or sys.modules.get(value.__name__) is not value
                )
                foreign = None
                if needs_clone and not collisions:
                    name = f"_b3_authenticated_{id(registry)}_clone_{len(registry.module_names)}"
                    foreign = reserve(name)
                result = register(cache, key, value)
                if foreign is not None:
                    assert sys.modules.get(name) is foreign
                return result

            cls.__setitem__ = occupied_clone

        observed = _observe_authenticated_class(monkeypatch, public, "_OwnedModules", install)
    try:
        with pytest.raises(ValueError):
            public.run(request, output)
        assert observed and len(collisions) == 1
        assert all(sys.modules.get(name) is foreign for name, foreign in collisions)
        _assert_no_complete_output(output)
    finally:
        for name, foreign in collisions:
            if sys.modules.get(name) is foreign:
                del sys.modules[name]
        _assert_no_owned_modules(before)

def _inject_public_acquisition_fstat(
    monkeypatch, public, output, stage, *, foreign_claim=False, ownership_probe=None, foreign_fd=False
):
    """Fail the first descriptor fstat after the actual targeted public open.

    Test cleanup releases an old RED implementation's leaked descriptor only
    if its still-live inode is exactly the one recorded by ordinary fstat.
    A foreign claim is a new entry while the original owned inode remains live.
    """
    ordinary_open, ordinary_fstat, ordinary_close = os.open, os.fstat, os.close
    ordinary_stat = os.stat
    method_name = {"claim": "claim", "staging": "prepare", "destination": "_destination"}[stage]
    method = getattr(public._Publication, method_name)
    active, pending, acquired, failures, foreign, reused = [], set(), [], [], [], []
    claim = output.with_name(output.name + ".claim")

    def during_method(owner, *args, **kwargs):
        active.append(True)
        try:
            return method(owner, *args, **kwargs)
        finally:
            active.pop()

    def record_open(path, flags, *args, **kwargs):
        fd = ordinary_open(path, flags, *args, **kwargs)
        name = Path(path).name
        target = (
            (stage == "claim" and name == claim.name and flags & os.O_CREAT)
            or (stage == "staging" and name == "publication" and flags & os.O_DIRECTORY)
            or (stage == "destination" and name == output.name and flags & os.O_DIRECTORY)
        )
        if active and target and not acquired:
            opened = ordinary_fstat(fd)
            acquired.append((fd, (opened.st_dev, opened.st_ino)))
            pending.add(fd)
        return fd

    def fail_admission(fd):
        if fd in pending and ownership_probe is None:
            pending.remove(fd)
            failures.append(fd)
            if foreign_claim:
                assert stage == "claim" and claim.exists()
                claim.unlink()
                claim.write_bytes(b"foreign writer claim after acquisition\n")
                current = claim.stat()
                foreign.append((current.st_dev, current.st_ino))
            if foreign_fd:
                assert stage == "claim"
                alternate = output.parent / "foreign-descriptor.txt"
                alternate.write_bytes(b"foreign descriptor stays live\n")
                ordinary_close(fd)
                replacement = ordinary_open(alternate, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
                if replacement != fd:
                    os.dup2(replacement, fd)
                    ordinary_close(replacement)
                current = ordinary_fstat(fd)
                reused.append((fd, (current.st_dev, current.st_ino)))
            raise OSError(errno.EIO, "injected acquisition fstat failure")
        return ordinary_fstat(fd)

    def fail_first_ownership_probe(path, *args, **kwargs):
        if ownership_probe is not None and pending:
            fd = next(iter(pending))
            matches = (ownership_probe == "descriptor" and type(path) is int and path == fd) or (
                ownership_probe == "proc" and path == f"/proc/self/fd/{fd}"
            )
            if matches:
                pending.remove(fd)
                failures.append(fd)
                raise OSError(errno.EIO, "injected ownership descriptor stat failure")
        return ordinary_stat(path, *args, **kwargs)

    monkeypatch.setattr(public._Publication, method_name, during_method)
    monkeypatch.setattr(os, "open", record_open)
    monkeypatch.setattr(os, "fstat", fail_admission)
    monkeypatch.setattr(os, "stat", fail_first_ownership_probe)

    def release_test_owned_red_leaks():
        for fd, identity in acquired:
            try:
                current = ordinary_fstat(fd)
            except OSError as error:
                assert error.errno == errno.EBADF
            else:
                if (current.st_dev, current.st_ino) == identity:
                    ordinary_close(fd)
        for fd, identity in reused:
            try:
                current = ordinary_fstat(fd)
            except OSError as error:
                assert error.errno == errno.EBADF
            else:
                if (current.st_dev, current.st_ino) == identity:
                    ordinary_close(fd)
        if foreign:
            try:
                current = claim.lstat()
            except FileNotFoundError:
                pass
            else:
                if (current.st_dev, current.st_ino) == foreign[0]:
                    claim.unlink()

    return SimpleNamespace(
        acquired=acquired,
        failures=failures,
        foreign=foreign,
        reused=reused,
        fstat=ordinary_fstat,
        cleanup=release_test_owned_red_leaks,
    )
